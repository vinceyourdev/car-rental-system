  <script>
    function loadTransactions() {
      const records = JSON.parse(localStorage.getItem('transactionRecords') || '[]');
      const list = document.getElementById('transaction-list');
      
      if (records.length === 0) {
        list.innerHTML = '<p class="empty-state"><i class="fas fa-inbox" style="font-size:1.5rem; opacity:0.5;"></i><br>No transactions yet.</p>';
        return;
      }
      
      list.innerHTML = records.map(tx => `
        <div class="transaction-item">
          <p><strong>Transaction ID:</strong> ${tx.transactionId}</p>
          <p><strong>Vehicle:</strong> ${tx.vehicle}</p>
          <p><strong>Name:</strong> ${tx.fullName}</p>
          <p><strong>Total Amount:</strong> ₱${Number(tx.totalAmount).toLocaleString('en-PH', {minimumFractionDigits:2})}</p>
          <p><strong>Date & Time:</strong> ${new Date(tx.timestamp).toLocaleString()}</p>
          <p><strong>Status:</strong> ✅ ${tx.status}</p>
        </div>
      `).join('');
    }

    // Clear data on logout
    document.getElementById('logout-link').addEventListener('click', function(e) {
      e.preventDefault();
      localStorage.removeItem('userName');
      window.location.href = 'index.html';
    });

    document.addEventListener('DOMContentLoaded', loadTransactions);
  </script>
